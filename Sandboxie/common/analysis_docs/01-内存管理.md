# 01 - 内存管理模块详解

## 模块概述

内存管理是 Sandboxie Common 库的核心基础设施，提供高效的内存分配和管理机制。

### 包含文件
- `pool.c / pool.h` - 内存池管理器
- `list.c / list.h` - 双向链表
- `lock.c / lock.h` - 线程同步锁

---

## 一、Pool 内存池管理器

### 1.1 设计目标

**为什么需要内存池？**

传统内存分配的问题：
- 系统调用开销大（ExAllocatePool/VirtualAlloc）
- 频繁分配导致内存碎片
- 小对象分配效率低
- 难以统计和管理内存使用

**内存池的优势：**
- ✅ 批量分配，减少系统调用
- ✅ 固定大小单元，避免碎片
- ✅ 快速分配和释放
- ✅ 统一管理，便于调试

### 1.2 核心数据结构

#### POOL 结构
```c
struct POOL {
    ULONG eyecatcher;              // 魔数：用于验证池的有效性
    
    // 同步机制（根据环境不同）
    #ifdef KERNEL_MODE
        PERESOURCE lock;           // 内核态：资源锁
    #else
        volatile LONG thread;      // 用户态：线程ID
        CRITICAL_SECTION lock;     // 用户态：临界区
    #endif
    
    LIST pages;                    // 可用页面链表
    LIST full_pages;               // 已满页面链表
    LIST large_chunks;             // 大块内存链表
    
    UCHAR initial_bitmap[PAGE_BITMAP_SIZE];  // 初始位图模板
};
```

#### PAGE 结构
```c
struct PAGE {
    LIST_ELEM list_elem;           // 链表节点
    PAGE *next;                    // 下一页指针
    POOL *pool;                    // 所属池
    ULONG eyecatcher;              // 魔数验证
    USHORT num_free;               // 空闲单元格数（估计值）
    
    // 紧随其后的内存布局：
    // [PAGE_HEADER] [BITMAP] [CELLS...]
};
```

#### LARGE_CHUNK 结构
```c
struct LARGE_CHUNK {
    LIST_ELEM list_elem;           // 链表节点
    ULONG eyecatcher;              // 魔数验证
    POOL *pool;                    // 所属池
    void *ptr;                     // 实际内存指针
    ULONG size;                    // 大小（页面对齐）
};
```

### 1.3 内存布局

#### 页面内存布局
```
┌─────────────────────────────────────────────────────────┐
│                    PAGE 结构 (32 bytes)                  │
├─────────────────────────────────────────────────────────┤
│                  位图 BITMAP (变长)                      │
│  每个 bit 表示一个 CELL 的占用状态                       │
│  0 = 空闲, 1 = 已占用                                   │
├─────────────────────────────────────────────────────────┤
│                   CELL 0 (16/128 bytes)                 │
├─────────────────────────────────────────────────────────┤
│                   CELL 1                                │
├─────────────────────────────────────────────────────────┤
│                   ...                                   │
├─────────────────────────────────────────────────────────┤
│                   CELL N                                │
└─────────────────────────────────────────────────────────┘

内核模式：
- 页面大小：4096 字节
- 单元格大小：16 字节
- 单元格数量：约 250 个

用户模式：
- 页面大小：65536 字节
- 单元格大小：128 字节
- 单元格数量：约 500 个
```

### 1.4 核心算法

#### 分配算法 (Pool_Alloc)

```c
void *Pool_Alloc(POOL *pool, ULONG size)
{
    // 1. 检查大小限制
    if (size >= LARGE_CHUNK_MAXIMUM)
        return NULL;
    
    // 2. 判断分配类型
    if (size > LARGE_CHUNK_MINIMUM) {
        // 大块分配：直接从系统申请
        return Pool_Get_Large_Chunk(pool, size);
    } else {
        // 小块分配：从页面中分配单元格
        ULONG num_cells = NUM_CELLS(size);
        return Pool_Get_Cells(pool, num_cells);
    }
}
```

#### 单元格分配算法 (Pool_Get_Cells)

```c
void *Pool_Get_Cells(POOL *pool, ULONG size)
{
    LOCK(pool);
    
    // 1. 遍历可用页面链表
    for (page in pool->pages) {
        if (page->num_free >= size) {
            // 2. 在页面中查找连续空闲单元格
            index = Pool_Find_Cells(page, size);
            if (index != -1) {
                // 3. 标记单元格为已占用
                mark_cells_used(page, index, size);
                
                // 4. 更新空闲计数
                page->num_free -= size;
                
                // 5. 如果页面几乎满了，移到 full_pages
                if (page->num_free < FULL_PAGE_THRESHOLD) {
                    move_to_full_pages(page);
                }
                
                UNLOCK(pool);
                return get_cell_address(page, index);
            }
            
            // 估计值不准确，更新它
            page->num_free = size - 1;
        }
    }
    
    // 6. 没有合适页面，分配新页面
    page = Pool_Alloc_Page(pool);
    mark_cells_used(page, 0, size);
    
    UNLOCK(pool);
    return get_cell_address(page, 0);
}
```

#### 位图查找算法 (Pool_Find_Cells)

```c
ULONG Pool_Find_Cells(PAGE *page, ULONG size)
{
    UCHAR *bitmap = get_bitmap(page);
    ULONG i = 0;
    
    while (i < NUM_PAGE_CELLS) {
        // 跳过已占用的单元格
        if (is_cell_used(bitmap, i)) {
            i = skip_used_cells(bitmap, i);
            continue;
        }
        
        // 检查是否有足够的连续空闲单元格
        ULONG j = i;
        while (j < NUM_PAGE_CELLS && j - i < size) {
            if (is_cell_used(bitmap, j))
                break;
            j++;
        }
        
        if (j - i >= size) {
            return i;  // 找到了！
        }
        
        i = j;
    }
    
    return -1;  // 未找到
}
```

### 1.5 性能优化技术

#### 优化1：分离满页面和可用页面

```c
// 问题：遍历所有页面效率低
// 解决：维护两个链表
LIST pages;       // 有足够空闲空间的页面
LIST full_pages;  // 几乎满的页面（不参与搜索）

// 阈值：少于4个空闲单元格就认为是"满"
#define FULL_PAGE_THRESHOLD 4
```

#### 优化2：估计空闲单元格数

```c
// 问题：每次都扫描位图太慢
// 解决：维护估计值
USHORT num_free;  // 估计的空闲单元格数

// 更新策略：
// - 分配时减少
// - 释放时增加
// - 如果估计值不准，调整为 size-1
```

#### 优化3：位图快速操作

```c
// 快速跳过8个单元格
if (mask == 1 && *bitmap == 0xFF) {
    j += 8;
    ++bitmap;
}

// 快速标记8个单元格
if (mask == 1 && size > 8) {
    *bitmap = 0xFF;
    ++bitmap;
    size -= 8;
}
```

#### 优化4：大块直接分配

```c
// 问题：大内存用单元格管理浪费
// 解决：超过阈值直接系统分配
#define LARGE_CHUNK_MINIMUM (POOL_PAGE_SIZE * 3 / 4)

if (size > LARGE_CHUNK_MINIMUM) {
    // 直接 VirtualAlloc/ExAllocatePool
    // 在块尾部存储 LARGE_CHUNK 结构
}
```

### 1.6 线程安全机制

#### 内核态锁

```c
PERESOURCE lock;

// 初始化
ExInitializeResourceLite(pool->lock);

// 加锁
KeRaiseIrql(APC_LEVEL, &irql);
ExAcquireResourceExclusiveLite(pool->lock, TRUE);

// 解锁
ExReleaseResourceLite(pool->lock);
KeLowerIrql(irql);
```

#### 用户态锁

```c
CRITICAL_SECTION lock;
volatile LONG thread;

// 初始化
InitializeCriticalSectionAndSpinCount(&pool->lock, 1000);

// 加锁（支持递归）
LONG tid = GetCurrentThreadId();
if (InterlockedCompareExchange(&pool->thread, tid, 0) != tid) {
    EnterCriticalSection(&pool->lock);
    pool->thread = tid;
}

// 解锁
LeaveCriticalSection(&pool->lock);
InterlockedExchange(&pool->thread, 0);
```

## 二、List 双向链表

### 2.1 数据结构

```c
typedef struct LIST_ELEM {
    struct LIST_ELEM *next;  // 后继节点
    struct LIST_ELEM *prev;  // 前驱节点
} LIST_ELEM;

typedef struct {
    LIST_ELEM *head;         // 链表头
    LIST_ELEM *tail;         // 链表尾
    int count;               // 元素计数
} LIST;
```

### 2.2 核心操作

#### 插入操作

```c
// 在 oldelem 之前插入 newelem
void List_Insert_Before(LIST *list, void *oldelem, void *newelem)
{
    LIST_ELEM *curr = (LIST_ELEM *)oldelem;
    LIST_ELEM *nelm = (LIST_ELEM *)newelem;
    
    ++list->count;
    
    if (curr == list->head || curr == NULL) {
        // 插入到头部
        nelm->prev = NULL;
        nelm->next = list->head;
        if (list->head)
            list->head->prev = nelm;
        else
            list->tail = nelm;  // 空链表
        list->head = nelm;
    } else {
        // 插入到中间
        LIST_ELEM *prev = curr->prev;
        prev->next = nelm;
        nelm->prev = prev;
        curr->prev = nelm;
        nelm->next = curr;
    }
}
```

#### 删除操作

```c
void List_Remove(LIST *list, void *elem)
{
    LIST_ELEM *curr = (LIST_ELEM *)elem;
    LIST_ELEM *prev = curr->prev;
    LIST_ELEM *next = curr->next;
    
    --list->count;
    
    if (prev == NULL && next == NULL) {
        // 最后一个元素
        list->head = NULL;
        list->tail = NULL;
    } else if (prev == NULL) {
        // 头节点
        list->head = next;
        next->prev = NULL;
    } else if (next == NULL) {
        // 尾节点
        list->tail = prev;
        prev->next = NULL;
    } else {
        // 中间节点
        prev->next = next;
        next->prev = prev;
    }
}
```

### 2.3 使用示例

```c
// 定义节点结构
typedef struct MY_NODE {
    LIST_ELEM elem;  // 必须是第一个成员
    int data;
} MY_NODE;

// 初始化链表
LIST mylist;
List_Init(&mylist);

// 插入节点
MY_NODE *node1 = malloc(sizeof(MY_NODE));
node1->data = 100;
List_Insert_Before(&mylist, NULL, node1);  // 插入到头部

// 遍历链表
MY_NODE *node = (MY_NODE *)List_Head(&mylist);
while (node) {
    printf("Data: %d\n", node->data);
    node = (MY_NODE *)List_Next(node);
}

// 删除节点
List_Remove(&mylist, node1);
free(node1);
```

## 三、Lock 锁机制

### 3.1 锁类型

```c
typedef ULONG LOCK;

#define LOCK_FREE 0
#define LOCK_EXCLUSIVE 1
#define LOCK_SHARED 2
```

### 3.2 核心函数

```c
// 获取独占锁
void Lock_Exclusive(LOCK *lock, const WCHAR *name);

// 获取共享锁
void Lock_Shared(LOCK *lock, const WCHAR *name);

// 释放锁
void Lock_Unlock(LOCK *lock, const WCHAR *name);
```

### 3.3 实现原理

使用原子操作实现自旋锁：

```c
void Lock_Exclusive(LOCK *lock, const WCHAR *name)
{
    while (InterlockedCompareExchange(lock, LOCK_EXCLUSIVE, LOCK_FREE) 
           != LOCK_FREE) {
        // 自旋等待
        YieldProcessor();
    }
}

void Lock_Unlock(LOCK *lock, const WCHAR *name)
{
    InterlockedExchange(lock, LOCK_FREE);
}
```

## 五、最佳实践

### 5.1 何时使用内存池

✅ **适合使用：**
- 频繁分配小对象（< 1KB）
- 对象生命周期相似
- 需要统一管理内存
- 性能敏感场景

❌ **不适合使用：**
- 偶尔分配大对象（> 10KB）
- 对象大小差异很大
- 长期持有内存
- 内存受限环境

### 5.2 使用建议

```c
// 1. 为每个模块创建独立的池
POOL *file_pool = Pool_Create();
POOL *registry_pool = Pool_Create();

// 2. 及时释放不需要的内存
Pool_Free(ptr, size);

// 3. 模块退出时删除池
Pool_Delete(file_pool);

// 4. 避免跨池操作
// 错误：用 pool1 分配，用 pool2 释放
void *ptr = Pool_Alloc(pool1, 100);
Pool_Free(ptr, 100);  // 会查找 ptr 所属的池

// 5. 记录分配大小
typedef struct {
    ULONG size;
    UCHAR data[1];
} ALLOC_HEADER;

void *my_alloc(POOL *pool, ULONG size) {
    ALLOC_HEADER *hdr = Pool_Alloc(pool, 
                                    sizeof(ALLOC_HEADER) + size);
    hdr->size = size;
    return hdr->data;
}
```
